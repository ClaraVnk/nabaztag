# A whole rabbit booting onto a WPA2 network and reaching for its warren.
#
# For ./rabbit-vm, not ./rabbit: this one runs the Metal VM, so the bytecode
# does the driving — it reads its configuration, scans, associates, runs the
# four-way handshake, asks for an address by DHCP, resolves its server and
# opens a connection to it. Pair it with an EEPROM whose SSID and cipher match
# the AP below, or the bytecode's own netAuth check refuses to dial:
#
#   python3 mkconf.py --ssid Terrier --psk nabaztag --security wpa2 \
#     --server warren.example:8080/vl -o rabbit.eeprom
#   ./rabbit-vm scenarios/boot-wpa2.txt --warren localhost

ap 02:00:5e:10:00:01 Terrier 6 wpa2-ccmp -52
psk nabaztag
tick 50
stop 90000

at 20000 expect RUN
